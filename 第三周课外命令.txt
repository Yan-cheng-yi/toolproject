import json
import sys
from pathlib import Path

TODO_FILE = Path.home() / ".todo_list.json"

def load_todos():
    if TODO_FILE.exists():
        with open(TODO_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return []

def save_todos(todos):
    with open(TODO_FILE, "w", encoding="utf-8") as f:
        json.dump(todos, f, ensure_ascii=False, indent=2)

def add_task(text):
    todos = load_todos()
    todos.append({"task": text, "done": False})
    save_todos(todos)
    print(f"已添加: {text}")

def list_tasks():
    todos = load_todos()
    if not todos:
        print("暂无任务")
        return
    for i, t in enumerate(todos, 1):
        mark = "[x]" if t["done"] else "[ ]"
        print(f"{i}. {mark} {t['task']}")

def done_task(index):
    todos = load_todos()
    if 1 <= index <= len(todos):
        todos[index - 1]["done"] = True
        save_todos(todos)
        print(f"已完成: {todos[index - 1]['task']}")
    else:
        print("序号不存在")

def delete_task(index):
    todos = load_todos()
    if 1 <= index <= len(todos):
        removed = todos.pop(index - 1)
        save_todos(todos)
        print(f"已删除: {removed['task']}")
    else:
        print("序号不存在")

def main():
    if len(sys.argv) < 2:
        print("用法: python todo.py <add|list|done|delete> [内容/序号]")
        return
    cmd = sys.argv[1]
    if cmd == "add":
        add_task(" ".join(sys.argv[2:]))
    elif cmd == "list":
        list_tasks()
    elif cmd == "done":
        done_task(int(sys.argv[2]))
    elif cmd == "delete":
        delete_task(int(sys.argv[2]))
    else:
        print(f"未知命令: {cmd}")

if __name__ == "__main__":
    main()
EOF
