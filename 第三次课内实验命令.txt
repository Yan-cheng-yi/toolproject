import torch
from torch import nn
torch.manual_seed(20260907)
x = torch.linspace(-1, 1, 101).reshape(-1, 1)
y = 3 * x - 1
loss_fn = nn.MSELoss()
model = nn.Linear(1, 1)
opt = torch.optim.SGD(model.parameters(), lr=0.1)
for _ in range(200):
    pred = model(x)
    loss = loss_fn(pred, y)
    opt.zero_grad()
    loss.backward()
    opt.step()
model.eval()
with torch.no_grad():
    pred_final = model(x)
    loss_final = loss_fn(pred_final, y)
    weight = model.weight.item()
    bias = model.bias.item()
    print(f"final loss: {loss_final.item():.6f}")
    print(f"weight: {weight:.4f}, bias: {bias:.4f}")
